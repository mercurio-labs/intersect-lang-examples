The language

Rcore defines the legal state transitions. Xtext turns start, wait and stop into command objects. Intersect Run schedules those commands in logical milliseconds and records the result.

These authored definitions are compiled by the same Rust language package used by the live example.

Metamodel

Structure and behavior in Rcore

stopwatch.rcore
Download ↓
rcore 1;
package stopwatch version "1";
metamodel "urn:intersect:example:stopwatch:1" prefix sw {
  class Script {
    attr name: String [1,1];
    ref commands: Command [0,*] containment;
    op nextState(state: String [1,1], command: String [1,1]) -> String [1,1];
  }
  class Command {
    attr kind: String [1,1];
    attr amount: Int32 [0,1];
    attr unit: String [0,1];
  }
}
behavior {
  import model "urn:intersect:example:stopwatch:1" as sw;
  // State machine authority: idle --start--> running --stop--> stopped.
  // Starting a stopped watch resumes its accumulated elapsed time.
  // Waiting never changes state. All other transitions are rejected.
  fn next_state(self:sw.Script, state:String, command:String)->String {
    if command == "wait" { state } else {
      if command == "start" {
        if state == "idle" || state == "stopped" { "running" } else { "invalid" }
      } else {
        if command == "stop" && state == "running" { "stopped" } else { "invalid" }
      }
    }
  }
}

Parser

Concrete syntax in Xtext

grammar example.Stopwatch with org.eclipse.xtext.common.Terminals
import "urn:intersect:example:stopwatch:1" as sw
import "http://www.eclipse.org/emf/2002/Ecore" as ecore
Script returns sw::Script: 'stopwatch' name=ID '{' commands+=Command* '}';
Command returns sw::Command: Start | Wait | Stop;
Start returns sw::Command: kind='start';
Stop returns sw::Command: kind='stop';
Wait returns sw::Command: kind='wait' amount=INT unit=Unit;
Unit returns ecore::EString: 'ms' | 's';

What to try

  • Start the watch, wait two seconds, then stop it.
  • Wait while stopped, then start again: elapsed time resumes rather than resetting.
  • Change a duration and rerun. Step through the trace to compare simulation time with elapsed time.

The sample ends at 3250 ms of simulation time with 2750 ms on the watch. The 500 ms spent stopped does not count.

fn main() -> Result<(), Box<dyn std::error::Error>> {
    // Parse authored commands, bind the Rcore state machine, execute through
    // Run's scheduler, then persist and reload its portable trace observations.
    let watch = examples_stopwatch::Stopwatch::load(examples_stopwatch::SAMPLE)?;
    let result = watch.execute()?;
    assert_eq!(result["summary"]["elapsed_ms"], 2750);
    std::fs::create_dir_all("target/runs")?;
    std::fs::write(
        "target/runs/stopwatch.json",
        serde_json::to_vec_pretty(&result)?,
    )?;
    let replay: serde_json::Value =
        serde_json::from_slice(&std::fs::read("target/runs/stopwatch.json")?)?;
    assert_eq!(result, replay);
    println!(
        "Stopped at 3250 ms of simulation time, with 2750 ms elapsed. Trace saved to target/runs/stopwatch.json"
    );
    Ok(())
}
