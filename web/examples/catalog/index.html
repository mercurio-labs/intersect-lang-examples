The language

The grammar constructs packages, entities and features. Rcore scope functions resolve names relative to packages, imports and qualified paths. The explorer uses the resulting frozen model to show each reference's actual target.

Catalog retains an Ecore metamodel. The Rcore below is generated by the same native migration API used during loading. Download the original Ecore.

Scoping and validation bindings (.rcore)
catalog-scoping.rcore
Download ↓
rcore 1;
package example.catalog.scoping version "1";
import model "urn:intersect:example:catalog:1" as c;

// Xtext's default scoping for this language, written in Rcore. A name is
// resolved as Xtext's ImportedNamespaceAwareLocalScopeProvider over the
// DefaultGlobalScopeProvider does: relative to each enclosing package, nearest
// first; then through the resource's imports (`import a.b.*` and
// `import a.b.C`); then as a fully qualified name among every loaded resource.
// An element's qualified name is its packages' names and its own, joined
// with ".".

bind scope c.Entity.superType = super_type;
bind scope c.Feature.type = feature_type;

fn super_type(r:c.Entity, name:String)->Sequence<c.Entity> =
  candidates(r, name).selectByKind::<c.Entity>();

fn feature_type(r:c.Feature, name:String)->Sequence<c.Type> =
  candidates(r, name).selectByKind::<c.Type>();

fn candidates(r:c.Element, name:String)->Sequence<c.Element> {
  let local = in_packages(r, ancestors::<c.Package>(r), name, 0);
  if local.notEmpty() { local } else {
    let imported = through_imports(r, name);
    if imported.notEmpty() { imported } else { with_qualified_name(r, name) }
  }
}

fn in_packages(r:c.Element, packages:OrderedSet<c.Package>, name:String, i:Int64)->Sequence<c.Element> =
  match packages.at(i) {
    Some(p) => {
      let found = with_qualified_name(r, qualified_name(p) + "." + name);
      if found.notEmpty() { found } else { in_packages(r, packages, name, i + 1) }
    },
    None => []
  };

fn through_imports(r:c.Element, name:String)->Sequence<c.Element> =
  ancestors::<c.Model>(r)
    .flatMap(|m| m.imports)
    .flatMap(|i| imported(r, i.importedNamespace, name));

// `a.b.*` imports every element directly in a.b; `a.b.C` imports C as C.
fn imported(r:c.Element, namespace:String, name:String)->Sequence<c.Element> {
  let wildcard = namespace.split(".*");
  if wildcard.size() == 2 {
    match wildcard.first() {
      Some(prefix) => with_qualified_name(r, prefix + "." + name),
      None => []
    }
  } else {
    if namespace.split(".").last() == Some(name) { with_qualified_name(r, namespace) } else { [] }
  }
}

fn with_qualified_name(r:c.Element, name:String)->Sequence<c.Element> =
  roots::<c.Model>(r)
    .flatMap(|m| all(Sequence(m.elements)))
    .filter(|e| qualified_name(e) == name);

fn all(elements:Sequence<c.Element>)->Sequence<c.Element> =
  elements.concat(elements.selectByKind::<c.Package>().flatMap(|p| all(Sequence(p.elements))));

fn qualified_name(e:c.Element)->String =
  match ancestors::<c.Package>(e).first() {
    Some(p) => qualified_name(p) + "." + e.name,
    None => e.name
  };

What to try

  • Rebuild the two bundled resources and inspect their objects.
  • Follow a type or superclass link to highlight the resolved target.
  • Change a resource and rebuild. Try an unresolved name to observe a linking error.

The sample resolves nine type and superclass references across the two resources. This is model exploration; the language has no timed firing behavior.

catalog_lifecycle.rs
Download ↓
//! Handwritten walkthrough: prepare -> parse two resources -> link -> validate
//! -> inspect references -> save and restore a compiled language pack.
use examples_support::Loaded;
use intersect_model::core::frozen::Obj;
use std::{collections::BTreeSet, fs, path::Path};

fn qualified(object: Obj<'_>) -> String {
    let mut current = Some(object);
    let mut names = Vec::new();
    while let Some(object) = current {
        if let Some(name) = object.feature("name").and_then(|id| object.str(id)) {
            names.push(name);
        }
        current = object.container().map(|(owner, _)| owner);
    }
    names.reverse();
    names.join(".")
}

fn references(loaded: &Loaded) -> Result<BTreeSet<(String, String, String)>, String> {
    if loaded.has_errors() {
        return Err(format!(
            "Catalog validation failed: {:?}",
            loaded.diagnostics
        ));
    }
    let mut references = BTreeSet::new();
    for object in loaded.model.objects() {
        for feature in ["type", "superType"] {
            if let Some(id) = object.feature(feature) {
                for target in object.refs(id) {
                    // Source identities, rather than display names, make the
                    // equality check sensitive to cross-resource targets.
                    references.insert((
                        object.identity().to_owned(),
                        feature.to_owned(),
                        target.identity().to_owned(),
                    ));
                    println!("  {}.{feature} -> {}", qualified(object), qualified(target));
                }
            }
        }
    }
    Ok(references)
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let output = Path::new("target/runs/catalog-lifecycle");
    fs::create_dir_all(output)?;

    // 1. Keep both resources in one load so the declared Rcore scopes can
    // resolve local, enclosing-package, imported and fully qualified names.
    let sources = examples_catalog::sample_sources();

    // 2. Preparation migrates the retained Ecore definitions to Rcore, compiles
    // the grammar and scopes, parses both sources, links and checks structure.
    let loaded = examples_support::load(&examples_catalog::LANGUAGE, &sources)?;
    println!(
        "{} resources, {} checked objects",
        sources.len(),
        loaded.model.objects().len()
    );

    // 3. Inspect the resolved model through frozen-engine reflection.
    let before = references(&loaded)?;
    if before.len() != 9 {
        return Err("Expected all nine sample reference targets".into());
    }

    // 4. Persist compiled definitions separately from model sources. Restore
    // only packs built by this host; the digest is not publisher authentication.
    let path = output.join("catalog.pack");
    fs::write(
        &path,
        examples_support::build_pack(&examples_catalog::LANGUAGE)?,
    )?;
    let restored =
        examples_support::load_pack(&examples_catalog::LANGUAGE, fs::read(&path)?, &sources)?;

    // 5. Reparse/link the sources with the restored definition. A pack must
    // preserve every reference's source and target identity, not just counts.
    let after = references(&restored)?;
    if before != after {
        return Err("Restoring the language pack changed reference identities".into());
    }
    println!(
        "All nine reference targets survived the round trip: {}",
        path.display()
    );
    Ok(())
}
